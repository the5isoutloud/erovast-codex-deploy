{{- /* Map of lower-case file name -> resource for every media file in the vault. Each file is fingerprinted: its URL contains a hash of its contents, e.g. Aldrich.1a2b3c….png, so replacing an image always gives it a new URL and browsers/CDNs can never show a stale copy. */ -}} {{- $s := newScratch -}} {{- $s.Set "idx" dict -}} {{- range resources.Match "vault/**" -}} {{- $k := partial "norm.html" (path.Base .Name) -}} {{- if not (index ($s.Get "idx") $k) -}}{{- $s.SetInMap "idx" $k (. | fingerprint "md5") -}}{{- end -}} {{- end -}} {{- return ($s.Get "idx") -}}
